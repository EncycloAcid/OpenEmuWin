cmake_minimum_required(VERSION 3.20)
project(OpenEmuWin LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

add_library(gambatte_host STATIC Core.cpp)
target_include_directories(gambatte_host PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(gambatte_host PRIVATE ${CMAKE_DL_LIBS})

if(WIN32)
    enable_language(RC)
    add_executable(OpenEmuWin WIN32 App.cpp Audio.cpp Game.cpp Storage.cpp OpenEmuWin.rc)
    target_link_libraries(OpenEmuWin PRIVATE gambatte_host bcrypt comctl32 comdlg32 gdi32 ole32 shell32 user32 uuid winmm xinput9_1_0)
    target_compile_definitions(OpenEmuWin PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
    target_compile_definitions(gambatte_host PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
    if(MSVC)
        target_link_options(OpenEmuWin PRIVATE /MANIFEST:NO)
    elseif(MINGW)
        target_link_options(OpenEmuWin PRIVATE -municode -static)
    endif()
    install(TARGETS OpenEmuWin RUNTIME DESTINATION .)
    install(FILES README.md DESTINATION .)
endif()

include(CTest)
set(GAMBATTE_SMOKE_CORE "" CACHE FILEPATH "Gambatte shared library for the smoke test")
if(BUILD_TESTING AND GAMBATTE_SMOKE_CORE)
    add_executable(core_smoke tests/CoreSmoke.cpp)
    target_link_libraries(core_smoke PRIVATE gambatte_host)
    if(MINGW)
        target_link_options(core_smoke PRIVATE -municode -static)
    endif()
    add_test(NAME gambatte_smoke COMMAND core_smoke "${GAMBATTE_SMOKE_CORE}")
endif()
